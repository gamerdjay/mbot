async function getNetworkData() { const btn = document.getElementById('btnFetch'); const res = document.getElementById('resultats'); btn.innerText = "Analyse..."; btn.disabled = true; // 1. Récupération des données Network Information API (Non supporté sur iOS/Safari) const conn = navigator.connection || navigator.mozConnection || navigator.webkitConnection; let networkType = "Non supporté par ce navigateur"; let rtt = "Non disponible"; let downlink = "Non disponible"; if (conn) { networkType = conn.effectiveType ? conn.effectiveType.toUpperCase() : "Inconnu"; rtt = conn.rtt ? conn.rtt + " ms" : "Inconnu"; downlink = conn.downlink ? conn.downlink + " Mb/s" : "Inconnu"; } // 2. Récupération des données réseau via une API sécurisée HTTPS (ipapi.co) let isp = "Recherche..."; let org = "Recherche..."; let ip = "Inconnue"; try { // En HTTPS pour fonctionner sur GitHub Pages const response = await fetch('https://ipapi.co/json/'); if (response.ok) { const data = await response.json(); isp = data.org || data.asn || "Non détecté"; org = data.network || data.version || "Non détecté"; ip = data.ip || "Inconnue"; } else { throw new Error("Erreur de réponse API"); } } catch (e) { isp = "Erreur de connexion (Bloqueur ou limite atteinte)"; org = "Erreur"; } // 3. Affichage des résultats res.innerHTML = `
> QUALITE ET TYPE DE RESEAU

Type de réseau estimé : ${networkType}

Latence estimée (RTT) : ${rtt}

Vitesse estimée (Downlink) : ${downlink}

> OPERATEUR ET IDENTIFICATION

Adresse IP : ${ip}

Opérateur (FAI / ISP) : ${isp}

Organisation réseau : ${org}

`; btn.innerText = "ACTUALISER"; btn.disabled = false; }
